Work bridge — 데이터 모델 (ERD)

18 테이블 · PostgreSQL(Neon) · Prisma 6 · 2026-10 · 건물주(client) ↔ 전문가(freelancer) 외주 매칭. 핵심: scope_items 한 행 = 계약 범위 조항 = 마일스톤 = 정산 단위

erDiagram
  users {
    uuid id PK
    varchar email UK
    varchar password_hash "소셜 전용은 null"
    varchar name
    user_role role "client | freelancer"
    varchar avatar_url
    text bio
    timestamptz created_at
  }
  oauth_accounts {
    uuid id PK
    uuid user_id FK
    varchar provider "google | kakao"
    varchar provider_user_id
    timestamptz linked_at
  }
  refresh_tokens {
    uuid id PK
    uuid user_id FK
    varchar token_hash "SHA-256, 평문 금지"
    timestamptz expires_at
    timestamptz revoked_at
  }
  client_profiles {
    uuid user_id PK, FK
    varchar building_name
    varchar building_type "상가|오피스|다세대|기타"
    varchar region
    varchar website
  }
  freelancer_profiles {
    uuid user_id PK, FK
    varchar main_category
    int years
    int hourly_rate
    varchar portfolio_url
  }
  skills {
    uuid id PK
    varchar name UK "기본 6종 + 커스텀"
    boolean is_custom
  }
  freelancer_skills {
    uuid freelancer_id PK, FK
    uuid skill_id PK, FK
  }
  project_skills {
    uuid project_id PK, FK
    uuid skill_id PK, FK
  }
  projects {
    uuid id PK
    uuid client_id FK
    varchar title
    text description
    varchar category
    timestamptz recruit_start_at
    timestamptz deadline_at
    int budget
    recruit_status recruit_status "scheduled|open|closed"
    deal_status deal_status "awaiting_contract..cancelled"
    timestamptz deleted_at "소프트 삭제"
    timestamptz created_at
  }
  applications {
    uuid id PK
    uuid project_id FK "project+freelancer unique"
    uuid freelancer_id FK
    text cover_letter
    int proposed_amount
    int proposed_days
    application_status status "pending|accepted|rejected"
    timestamptz created_at
  }
  scope_analyses {
    uuid id PK
    uuid project_id FK
    int total_estimate
    varchar model "사용한 LLM"
    timestamptz created_at
  }
  scope_items {
    uuid id PK
    uuid analysis_id FK
    int sort_order
    varchar name "예: 방수 재시공"
    text description
    int estimate
    text rationale "산정 근거"
  }
  contracts {
    uuid id PK
    uuid project_id FK
    uuid application_id FK
    int agreed_amount
    jsonb scope_snapshot "체결 시점 공정표 사본"
    timestamptz client_signed_at
    timestamptz freelancer_signed_at
    timestamptz created_at
  }
  milestones {
    uuid id PK
    uuid contract_id FK
    uuid scope_item_id FK
    int sort_order
    varchar title
    int amount
    milestone_status status "todo|in_progress|submitted|approved"
    int fee "수수료 10%"
    int net "실지급액"
    timestamptz approved_at
  }
  payments {
    uuid id PK
    uuid contract_id FK
    varchar pg_order_id UK
    varchar pg_payment_key
    int amount
    payment_status status "pending|paid|..|refunded"
    timestamptz paid_at
  }
  reviews {
    uuid id PK
    uuid project_id FK "project+direction unique"
    uuid author_id FK
    review_direction direction
    int rating "1-5"
    text comment
    varchar_arr tags "고정 태그 세트"
    timestamptz created_at
  }
  notifications {
    uuid id PK
    uuid user_id FK
    varchar type
    varchar title
    varchar body
    varchar link "클릭 시 이동"
    timestamptz read_at
    timestamptz created_at
  }
  bookmarks {
    uuid user_id PK, FK
    uuid project_id PK, FK
    timestamptz created_at
  }
